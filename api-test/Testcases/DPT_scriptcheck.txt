-- =============== C1: PERSONAL - SINGLE ===============
-- Declare parameters
DECLARE @CustomerCode NVARCHAR(10) = '11000001';
DECLARE @Categories NVARCHAR(50) = 'C1';
DECLARE @DepositAccount NVARCHAR(50) = '120077856632';
DECLARE @CatalogCode NVARCHAR(50) = 'CAEUR0000';
DECLARE @BranchCode NVARCHAR(10) = '007';
DECLARE @CurrencyCode NVARCHAR(10) = '03';
-- Query 1: SingleCustomer
SELECT Categories, * FROM [27.3.1.199].o9customer.dbo.SingleCustomer WHERE CustomerCode = @CustomerCode;
-- Query 2: DepositAccount
SELECT * FROM [27.3.1.199].o9deposit.dbo.DepositAccount WHERE AccountNumber = @DepositAccount;
-- Query 3: DepositExtensionAccountOfGroupDefinition
SELECT * FROM [27.3.1.199].o9deposit.dbo.DepositExtensionAccountOfGroupDefinition WHERE CatalogCode = @CatalogCode AND Categories = @Categories;
-- Query 4: DepositAccountGLs
SELECT * FROM [27.3.1.199].o9deposit.dbo.DepositAccountGLs WHERE DepositAccount = @DepositAccount;
-- Comparison Query
SELECT 
    CASE 
        WHEN (SELECT GLAccount 
              FROM [27.3.1.199].o9deposit.dbo.DepositAccountGLs 
              WHERE DepositAccount = @DepositAccount) = 
             (SELECT CONCAT(@BranchCode, ReplaceBy, @CurrencyCode) 
              FROM [27.3.1.199].o9deposit.dbo.DepositExtensionAccountOfGroupDefinition 
              WHERE CatalogCode = @CatalogCode AND Categories = @Categories)
        THEN 'The values are equal'
        ELSE 'The values are not equal'
    END AS ComparisonResult;

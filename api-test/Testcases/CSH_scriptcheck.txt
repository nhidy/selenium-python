-- Check cash - gl
select a.branchcode, a.balance + b.balance amountdiff from (
select branchcode, SUM(balance) balance, currencycode 
from o9accounting.dbo.AccountBalance where SUBSTRING(AccountNumber, 4, 13) = '1010301000101' 
group by branchcode, currencycode) a inner join 
(
select b.BranchCode, a.currencycode, SUM(CurrentBalance) balance from o9cash.[dbo].[CashList] a
  inner join o9admin.dbo.UserAccount b on a.CashierCode = b.UserCode
group by  b.BranchCode, a.currencycode) b on a.branchcode = b.branchcode and a.currencycode = b.currencycode
where a.balance + b.balance != 0;

select b.AccountNumber, b.Balance - a.DPTBALANCE diff from 
(select  SUM(CurrentBalance) DPTBALANCE, b.GLAccount
from o9deposit.dbo.depositAccount a
	inner join o9deposit.dbo.depositAccountGLs b on a.AccountNumber = b.DepositAccount
where CurrentBalance > 0
group by b.GLAccount) a
inner join o9Accounting.dbo.AccountBalance b on a.GLAccount=b.AccountNumber
where b.Balance != a.DPTBALANCE;

-- check master detail với OD 

select d.Balance, e.*, f.AccountName, Balance+TotalCredit from o9accounting.dbo.AccountBalance d 
inner join o9accounting.dbo.AccountChart f on d.AccountNumber = f.AccountNumber
inner join (select GLAccount, sum (OverDraftBalance) TotalCredit from 
(select a.*, 
case b.ClassificationStatus
when 'N' then 'CREDIT0' 
when 'P' then 'CREDIT1'
when 'O' then 'CREDIT2'
when 'D' then 'CREDIT3'
when 'L' then 'CREDIT4'
end NPLGroup, b.OverDraftBalance
from o9deposit.dbo.OverdraftContractGLs a left join o9deposit.dbo.OverdraftContract b on a.ODContract = b.ContractNumber) c
where NPLGroup = SysAccountName
group by  GLAccount) e on d.AccountNumber = e.GLAccount
where Balance + TotalCredit != 0;

SELECT '' DepositTransDetails, b.TransactionNumber, b.RefId, b.TransactionCode, b.TransactionDate, b.[Status], a.Entity, a.FieldName, a.EntityId, SUM(CASE WHEN a.[Status] = 'N' THEN 1 ELSE 0 END) CountSuccess, SUM(CASE WHEN a.[Status] <> 'N' THEN 1 ELSE 0 END) CountFail
FROM o9deposit.dbo.TransactionDetails a
JOIN
(
	SELECT DISTINCT a.TransactionNumber, a.RefId, a.TransactionCode, a.TransactionDate, a.[Status] FROM o9admin.dbo.TransactionJournalDone a
	JOIN o9deposit.dbo.TransactionDetails b ON a.RefId = b.RefId
	WHERE a.[Status] in ('C','N') AND b.[Status] <> 'N' --AND b.RefId NOT IN (SELECT RefId FROM @ignoreDptDetails)
) b ON a.RefId =  b.RefId
GROUP BY b.TransactionNumber, b.RefId, b.TransactionCode, b.TransactionDate, b.[Status], a.Entity, a.FieldName, a.EntityId
HAVING SUM(CASE WHEN a.[Status] = 'N' THEN 1 ELSE 0 END) = 0 AND b.TransactionDate >= '2023-10-19' ORDER BY b.TransactionDate

SELECT CASE WHEN b.AccountNumber = '002202030201010101' THEN 'Remember to subtract 150k' END DepositMasterDetails,
b.AccountNumber, c.AccountName, a.Balance - b.Balance [Deposit - Accounting]
FROM 
(
	SELECT b.GLAccount, SUM(a.CurrentBalance + ISNULL(c.OverDraftBalance,0)) Balance
	FROM o9deposit.dbo.DepositAccount a JOIN o9deposit.dbo.DepositAccountGLs b ON a.AccountNumber = b.DepositAccount LEFT JOIN
	(
		SELECT AccountNumber, OverDraftBalance FROM o9deposit.dbo.OverdraftContract WHERE ContractStatus = 0
	) c ON a.AccountNumber = c.AccountNumber
	GROUP BY b.GLAccount
) a
JOIN o9Accounting.dbo.AccountBalance b ON a.GLAccount = b.AccountNumber
JOIN o9Accounting.dbo.AccountChart c ON b.AccountNumber = c.AccountNumber
WHERE a.Balance != b.Balance
